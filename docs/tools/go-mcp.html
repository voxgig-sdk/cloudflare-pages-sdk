Tools

cloudflare-pages_list

List records for an entity.

{
  "type": "object",
  "required": [
    "entity"
  ],
  "properties": {
    "entity": {
      "type": "string",
      "description": "pages_deployment | pages_domain | pages_pages_assets_check_missing | pages_project | upload | upsert_hash"
    },
    "query": {
      "type": "object",
      "description": "Optional match fields for the selected entity."
    }
  }
}

Supported entities: pages_deployment, pages_domain, pages_project.

  }
}

Supported entities: pages_deployment, pages_domain, pages_project.
